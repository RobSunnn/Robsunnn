/**

 * Focused on building clean, scalable, cloud-native software.
 */
public final class Roberto extends Developer {

    private static final String LOCATION = "Bulgaria 🇧🇬";

    private final List<String> currentFocus;
    private final boolean openToCollaborate;

    public Roberto() {
        this.currentFocus = List.of(
            "Backend Development with Spring Boot 🖥️",
            "Cloud-Native Architecture (AWS) ☁️",  // Primary cloud provider
            "Shipping clean, scalable software 🔧"
        );
        this.openToCollaborate = true;
    }

    public static String getLocation() { return LOCATION; }
    public List<String> getCurrentFocus() { return Collections.unmodifiableList(currentFocus); }
    public boolean isOpenToCollaborate() { return openToCollaborate; }

    @Override
    public String introduce() {
        return String.format(
            "Passionate developer from %s who loves clean code, " +
            "scalable systems, and turning ideas into reality.",
            LOCATION

    }
